🛠️ 数据库建表 SQL (首次使用必读)

如果提示“数据表尚未初始化”，只需复制下方 SQL，粘贴至 Supabase 控制台的 SQL Editor 中点击 Run 即可完成一键建表： 

-- 1. 创建单词主表
create table if not exists public.words (
  id text primary key,
  word text not null,
  phonetic text default '',
  "partOfSpeech" text default 'n.',
  definition text default '',
  "exampleEn" text default '',
  "exampleCn" text default '',
  tags jsonb default '[]'::jsonb,
  status text default 'new',
  "reviewCount" integer default 0,
  interval integer default 0,
  "nextReviewDate" bigint,
  "lastReviewedAt" bigint,
  "userNotes" text default '',
  "createdAt" bigint,
  "lastModifiedAt" bigint
);

-- 2. 创建打卡统计表
create table if not exists public.study_stats (
  id text primary key default 'global_user_stats',
  "dailyCounts" jsonb default '{}'::jsonb,
  "checkInDates" jsonb default '[]'::jsonb,
  "bestStreak" integer default 0,
  "lastActiveDate" text,
  "updatedAt" bigint
);

-- 3. 开启权限策略
alter table public.words enable row level security;
alter table public.study_stats enable row level security;
create policy "Allow public words" on public.words for all using (true) with check (true);
create policy "Allow public stats" on public.study_stats for all using (true) with check (true);
